from urllib.parse import urlparse

def domain_kontrol(url, url_listesi):
    parsed_url = urlparse(url)
    girilen_domain = parsed_url.netloc
    
    for listedeki_url in url_listesi:
        parsed_listedeki_url = urlparse(listedeki_url)
        listedeki_domain = parsed_listedeki_url.netloc
        
        if girilen_domain == listedeki_domain:
            return True
    
    return False
